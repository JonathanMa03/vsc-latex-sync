\section{Example}

\begin{definition}
A random variable $X$ is Gaussian if $X \sim \Normal(\mu,\sigma^2)$.
\end{definition}

\begin{theorem}
Let $X_1,\dots,X_n \iid \Normal(\mu,\sigma^2)$. Then $\bar X$ is unbiased.
\end{theorem}

\begin{proof}
By linearity,
$
\E[\bar X]
=
\mu.
$
\end{proof}

\section{Algorithm Example}

\begin{algorithm}
\caption{Gradient Descent}
\begin{algorithmic}[1]
\State Initialize $x_0$
\For{$k=0,1,2,\dots$}
    \State $x_{k+1} \gets x_k - \eta \grad f(x_k)$
\EndFor
\end{algorithmic}
\end{algorithm}

\section{Code Example}

\begin{lstlisting}[language=Python]
import numpy as np
import pandas as pd
import scipy as 
def gradient_descent(x0, eta, grad, steps):
    x = x0
    for _ in range(steps):
        x = x - eta * grad(x)
    return x
